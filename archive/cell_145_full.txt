
# Phase 9.5: Advanced Regression with Classification Features

# ============================================================================
# IMPORTS AND PREREQUISITES
# ============================================================================
# Note: All required imports are centralized at the beginning of the notebook
# Phase 9.5 functions from finance_ml.advanced_models:
#   - prepare_regression_data
#   - create_classification_interactions
#   - compare_regressors

# Standard libraries (already imported at notebook start)
# import pandas as pd
# import numpy as np
# import matplotlib.pyplot as plt
# import seaborn as sns
# from pathlib import Path

# Sklearn metrics needed for Phase 9.5
from sklearn.metrics import r2_score, mean_absolute_error, mean_squared_error

# Import config and logger from finance_ml
from finance_ml.config import get_config
from finance_ml.logging_config import get_logger

# Get the global configuration instance (initialized earlier in notebook)
config = get_config()
logger = get_logger(__name__)

# ============================================================================
# CONSTANTS
# ============================================================================
PHASE_NAME = "PHASE 9.5 — SECTOR-OPTIMIZED REGRESSION MODELS"
TARGET_COL_DEFAULT = 'price_target'
TARGET_COL_FALLBACK = 'last_price'
TEST_SIZE_RATIO = 0.2
CV_FOLDS = 5
RANDOM_STATE = 42
QUANTILES = [0.1, 0.5, 0.9]
MIN_SECTOR_SAMPLES = 20
SAMPLE_SIZE_VISUALIZATION = 50
SEPARATOR_LINE = "=" * 80


# ============================================================================
# HELPER FUNCTIONS
# ============================================================================
def print_phase_header(phase_name: str) -> None:
    """Print formatted phase header."""
    print(f"\n{SEPARATOR_LINE}")
    print(phase_name)
    print(SEPARATOR_LINE)


def print_section_header(section_name: str) -> None:
    """Print formatted section header."""
    print(f"\n{SEPARATOR_LINE}")
    print(section_name)
    print(SEPARATOR_LINE)


def initialize_regression_dataset(all_stocks_phase94: pd.DataFrame) -> pd.DataFrame:
    """Initialize regression dataset from Phase 9.4 data."""
    regression_df = all_stocks_phase94.copy()
    print(f"\n📊 Starting dataset: {regression_df.shape}")
    print("   Includes classification meta-features from Phase 9.4")
    return regression_df


def prepare_target_variable(df: pd.DataFrame, target_col: str, fallback_col: str) -> tuple:
    """Prepare target variable, using fallback if needed."""
    if target_col not in df.columns:
        print(f"\n⚠ Warning: '{target_col}' not found. Using '{fallback_col}' as proxy target.")
        target_col = fallback_col

    df_clean = df.dropna(subset=[target_col])
    print(f"\n✓ Clean dataset: {df_clean.shape[0]:,} rows")
    return df_clean, target_col


# ============================================================================
# MAIN EXECUTION
# ============================================================================
print_phase_header(PHASE_NAME)

try:
    # Check if all_stocks_phase94 exists with more informative error
    if 'all_stocks_phase94' not in globals():
        raise NameError(
                "all_stocks_phase94 not found. Please run Phase 9.4 (Multi-Class Classification) first.\n"
                "Phase 9.4 creates classification meta-features required for regression models.\n"
                "Required notebook cells: Phase 9.4 sections 9.4.1 through 9.4.3"
                )

    # Validate that the variable is not None and is a DataFrame
    if all_stocks_phase94 is None:
        raise ValueError("all_stocks_phase94 is None. Phase 9.4 may have failed.")

    if not isinstance(all_stocks_phase94, pd.DataFrame):
        raise TypeError(
                f"all_stocks_phase94 must be a DataFrame, got {type(all_stocks_phase94).__name__}"
                )

    if len(all_stocks_phase94) == 0:
        raise ValueError("all_stocks_phase94 is empty. No data available for regression.")

    # Initialize dataset
    regression_df = initialize_regression_dataset(all_stocks_phase94)
    regression_df_clean, target_col = prepare_target_variable(
            regression_df, TARGET_COL_DEFAULT, TARGET_COL_FALLBACK
            )

    #%% md
    ### 9.5.1 Create Classification Interaction Features

    #%%
    print_section_header("9.5.1 — Creating Classification Interaction Features")

    # Extract classification and valuation columns
    classification_cols = [c for c in regression_df_clean.columns if c.startswith('event_prob_')]
    valuation_cols = [c for c in ['p_e', 'p_b', 'ev_ebitda', 'market_cap']
                      if c in regression_df_clean.columns]

    if classification_cols and valuation_cols:
        print(f"\nClassification features: {len(classification_cols)}")
        print(f"Valuation features: {len(valuation_cols)}")

        # Check for data quality issues
        print("\n🔧 Checking for zero/infinite values...")
        for col in valuation_cols:
            if col in regression_df_clean.columns:
                zero_count = (regression_df_clean[col] == 0).sum()
                inf_count = np.isinf(regression_df_clean[col]).sum()
                if zero_count > 0 or inf_count > 0:
                    print(f"  Warning: {col} has {zero_count} zeros and {inf_count} inf values")

        # Create interaction features
        regression_df_enhanced = create_classification_interactions(
                regression_df_clean,
                classification_cols=classification_cols,
                valuation_cols=valuation_cols
                )

        # Report results
        interaction_cols = [c for c in regression_df_enhanced.columns
                            if '_x_' in c and c not in regression_df_clean.columns]
        print(f"\n✓ Created {len(interaction_cols)} interaction features")
        if interaction_cols[:3]:
            print(f"  Examples: {', '.join(interaction_cols[:3])}")
    else:
        print("\n⚠ Skipping interaction features - missing required columns")
        regression_df_enhanced = regression_df_clean.copy()

    # Handle missing values (consolidated approach)
    print("\n🔧 Handling missing values...")
    nan_before = regression_df_enhanced.isnull().sum().sum()
    print(f"  NaN values before cleanup: {nan_before}")

    if nan_before > 0:
        # Single-pass cleanup for efficiency
        numeric_cols = regression_df_enhanced.select_dtypes(include=[np.number]).columns

        # Step 1: Replace inf with NaN
        regression_df_enhanced = regression_df_enhanced.replace([np.inf, -np.inf], np.nan)

        # Step 2: Fill numeric columns with median (excluding target)
        for col in numeric_cols:
            if col != target_col and regression_df_enhanced[col].isnull().any():
                median_val = regression_df_enhanced[col].median()
                fill_value = median_val if pd.notna(median_val) else 0
                regression_df_enhanced[col] = regression_df_enhanced[col].fillna(fill_value)

        # Step 3: Final safety - fill any remaining NaNs with 0
        regression_df_enhanced = regression_df_enhanced.fillna(0)

        nan_after = regression_df_enhanced.isnull().sum().sum()
        print(f"  NaN values after cleanup: {nan_after}")

        if nan_after == 0:
            print("✓ No NaNs remaining - data ready for modeling")
        else:
            print(f"⚠ Warning: {nan_after} NaN values remain after cleanup")

    #%% md
    ### 9.5.2 Prepare Regression Data

    #%%
    print_section_header("9.5.2 — Preparing Regression Data")

    X_train_reg, X_test_reg, y_train_reg, y_test_reg, feature_info = prepare_regression_data(
            regression_df_enhanced,
            target_col=target_col,
            test_size=TEST_SIZE_RATIO,
            random_state=RANDOM_STATE
            )

    print(f"\n✓ Data prepared:")
    print(f"  Train set: {X_train_reg.shape}")
    print(f"  Test set: {X_test_reg.shape}")
    print(f"  Numeric features: {len(feature_info.get('numeric_features', []))}")
    print(f"  Categorical features: {len(feature_info.get('categorical_features', []))}")

    #%% md
    ### 9.5.3 Compare Multiple Regression Models

    #%%
    print_section_header("9.5.3 — Comparing Multiple Regression Models")

    try:
        comparison_results = compare_regressors(
                X_train_reg, y_train_reg,
                test_size=TEST_SIZE_RATIO,
                cv=CV_FOLDS,
                random_state=RANDOM_STATE,
                ensure_nonnegative=True
                )

        results_df = pd.DataFrame(comparison_results).T.sort_values('r2', ascending=False)
        print("\n📊 Model Comparison Results:")
        print(results_df.to_string())

        # Visualize comparison only if we have valid results
        if not results_df.empty and len(results_df) > 0:
            fig, axes = plt.subplots(1, 3, figsize=(18, 5))

            results_df['r2'].plot(kind='barh', ax=axes[0], color='skyblue', edgecolor='black')
            axes[0].set_title('R² Score (higher is better)', fontweight='bold')
            axes[0].set_xlabel('R²')
            axes[0].set_xlim(0, 1.0)

            results_df['mae'].plot(kind='barh', ax=axes[1], color='coral', edgecolor='black')
            axes[1].set_title('Mean Absolute Error (lower is better)', fontweight='bold')
            axes[1].set_xlabel('MAE')

            results_df['train_time'].plot(kind='barh', ax=axes[2], color='lightgreen', edgecolor='black')
            axes[2].set_title('Training Time (seconds)', fontweight='bold')
            axes[2].set_xlabel('Time (s)')

            plt.tight_layout()
            plt.show()

            best_model_name = results_df.index[0]
            print(f"\n🏆 Best Model: {best_model_name}")
            print(f"   R²: {results_df.loc[best_model_name, 'r2']:.4f}")
            print(f"   MAE: {results_df.loc[best_model_name, 'mae']:.2f}")
        else:
            print("  ⚠ No models were successfully trained for comparison")
            best_model_name = "None"

    except ValueError as e:
        logger.warning(f"Model comparison failed due to data issues: {e}")
        print(f"\n⚠ Model comparison failed: {e}")
        print("  Check data quality and feature availability")
        results_df = pd.DataFrame()
        best_model_name = "None"
    except RuntimeError as e:
        logger.warning(f"Model comparison runtime error: {e}")
        print(f"\n⚠ Model comparison failed: {e}")
        results_df = pd.DataFrame()
        best_model_name = "None"
    except Exception as e:
        logger.error(f"Unexpected error in model comparison: {e}", exc_info=True)
        print(f"\n⚠ Model comparison failed: {e}")
        results_df = pd.DataFrame()
        best_model_name = "None"

    #%% md
    ### 9.5.4 Train Stacking Ensemble for Best Performance

    #%%
    print_section_header("9.5.4 — Training Stacking Ensemble")

    stacking_model, stacking_results = train_stacking_regressor(
            X_train_reg, y_train_reg, cv=CV_FOLDS, ensure_nonnegative=True
            )

    print(f"\n✓ Stacking Ensemble Trained:")
    print(f"  Base models: {', '.join(stacking_results.get('base_models', []))}")
    print(f"  Meta-learner: {stacking_results.get('meta_model', 'Unknown')}")
    print(f"  Train R²: {stacking_results.get('train_score', 0):.4f}")
    print(f"  CV R² (mean ± std): {stacking_results.get('cv_score', 0):.4f} ± {stacking_results.get('cv_std', 0):.4f}")

    # Test set predictions
    y_pred_stacking = stacking_model.predict(X_test_reg)

    test_metrics = {
        'mae': mean_absolute_error(y_test_reg, y_pred_stacking),
        'rmse': np.sqrt(mean_squared_error(y_test_reg, y_pred_stacking)),
        'r2': r2_score(y_test_reg, y_pred_stacking)
        }

    print(f"\n📊 Test Set Performance:")
    print(f"  MAE: {test_metrics['mae']:.2f}")
    print(f"  RMSE: {test_metrics['rmse']:.2f}")
    print(f"  R²: {test_metrics['r2']:.4f}")

    #%% md
    ### 9.5.5 Quantile Regression for Prediction Intervals

    #%%
    print_section_header("9.5.5 — Quantile Regression for Uncertainty Estimation")

    quantile_models, quantile_results = train_quantile_regressor(
            X_train_reg, y_train_reg, quantiles=QUANTILES
            )

    print(f"\n✓ Quantile Models Trained:")
    print(f"  Quantiles: {QUANTILES}")
    print(f"  Models: {len(quantile_models)}")

    # Generate predictions for each quantile
    predictions_quantile = {}
    for q, model in zip(QUANTILES, quantile_models):
        predictions_quantile[q] = model.predict(X_test_reg)

        # Calculate score safely with specific exception handling
        try:
            score = model.score(X_train_reg, y_train_reg)
            print(f"  Q{q}: {score:.4f} (train R²)")
        except AttributeError as e:
            # Model doesn't support score() method
            print(f"  Q{q}: Model trained successfully (score method not available)")
        except ValueError as e:
            # Invalid input data
            print(f"  Q{q}: Score calculation failed - {str(e)}")
        except Exception as e:
            # Other unexpected errors
            logger.warning(f"Unexpected error calculating score for Q{q}: {e}")
            print(f"  Q{q}: Model trained successfully (score calculation failed)")

    # Visualize prediction intervals with validation
    predictions_df = pd.DataFrame({
        'actual': y_test_reg.values,
        'lower_10': predictions_quantile[0.1],
        'median': predictions_quantile[0.5],
        'upper_90': predictions_quantile[0.9],
        'point_prediction': y_pred_stacking
        })

    # Validate quantile order (lower < median < upper)
    invalid_intervals = (
            (predictions_df['lower_10'] > predictions_df['median']) |
            (predictions_df['median'] > predictions_df['upper_90'])
    ).sum()

    if invalid_intervals > 0:
        print(f"  ⚠ Warning: {invalid_intervals} samples have invalid quantile order")
        print(f"    Correcting by sorting quantiles...")

        # Fix invalid intervals
        for idx in predictions_df.index:
            values = [
                predictions_df.loc[idx, 'lower_10'],
                predictions_df.loc[idx, 'median'],
                predictions_df.loc[idx, 'upper_90']
                ]
            sorted_values = sorted(values)
            predictions_df.loc[idx, 'lower_10'] = sorted_values[0]
            predictions_df.loc[idx, 'median'] = sorted_values[1]
            predictions_df.loc[idx, 'upper_90'] = sorted_values[2]

    print(f"\n📊 Prediction Intervals (first 5 samples):")
    print(predictions_df.head().to_string())

    # Plot prediction intervals
    sample_size = min(SAMPLE_SIZE_VISUALIZATION, len(predictions_df))
    sample_indices = range(sample_size)

    plt.figure(figsize=(14, 6))
    plt.fill_between(
            sample_indices,
            predictions_df['lower_10'][:sample_size],
            predictions_df['upper_90'][:sample_size],
            alpha=0.3, color='lightblue', label='80% Prediction Interval'
            )
    plt.plot(sample_indices, predictions_df['actual'][:sample_size],
             'o', color='black', label='Actual', markersize=4)
    plt.plot(sample_indices, predictions_df['median'][:sample_size],
             '-', color='blue', label='Median Prediction', linewidth=2)
    plt.plot(sample_indices, predictions_df['point_prediction'][:sample_size],
             '--', color='red', label='Stacking Prediction', linewidth=1.5)
    plt.xlabel('Sample Index')
    plt.ylabel('Price Target')
    plt.title('Prediction Intervals: Quantile Regression with Stacking Ensemble', fontweight='bold')
    plt.legend(loc='best')
    plt.grid(True, alpha=0.3)
    plt.tight_layout()
    plt.show()

    #%% md
    ### 9.5.6 Sector-Specific Models (Optional)

    #%%
    print_section_header("9.5.6 — Sector-Specific Model Training")

    if 'sector' in regression_df_enhanced.columns:
        feature_cols = list(X_train_reg.columns)

        models, sector_results = train_sector_specific_models(
                regression_df_enhanced,
                feature_cols=feature_cols,
                target_col=target_col,
                sector_col='sector',
                model_type='random_forest',
                min_samples=MIN_SECTOR_SAMPLES
                )

        print(f"\n✓ Sector-Specific Models Trained:")
        print(f"  Total sectors: {len(models)}")

        # Format and display results
        sector_metrics = sector_results.get('sector_metrics', sector_results)
        sector_summary = pd.DataFrame(sector_metrics).T

        # Determine best metric for sorting
        sort_metric = 'r2' if 'r2' in sector_summary.columns else 'train_score'
        if sort_metric in sector_summary.columns:
            sector_summary = sector_summary.sort_values(sort_metric, ascending=False)

            print(f"\n📊 Sector Model Performance (sorted by {sort_metric}):")
            display_cols = [c for c in ['train_score', 'r2', 'mae', 'rmse', 'model_type']
                            if c in sector_summary.columns]
            if display_cols:
                print(sector_summary[display_cols].head(10).to_string())
    else:
        print("\n⚠ Sector column not found - skipping sector-specific models")

    #%% md
    ### 9.5.7 Model Persistence

    #%%
    print_section_header("9.5.7 — Model Persistence")

    # Prepare models directory
    models_dir = Path(config.output_dir) / 'models'
    models_dir.mkdir(exist_ok=True, parents=True)

    # Create comprehensive stacking_results dict
    # FIX: Use results from train_stacking_regressor instead of calling cross_val_score again
    from datetime import datetime

    complete_stacking_results = {
        'train_score': stacking_results.get('train_score', 0),
        'cv_score': stacking_results.get('cv_score', 0),
        'cv_std': stacking_results.get('cv_std', 0),
        'test_score': test_metrics['r2'],
        'base_models': stacking_results.get('base_models', []),
        'meta_model': stacking_results.get('meta_model', 'Unknown'),
        'model': stacking_model
        }

    # Save stacking model
    stacking_metadata = {
        'model_type': 'stacking_ensemble',
        'features': list(X_train_reg.columns),
        'target': target_col,
        'date_trained': datetime.now().strftime('%Y-%m-%d'),
        'phase': '9.5',
        'train_score': complete_stacking_results['train_score'],
        'cv_score': complete_stacking_results['cv_score'],
        'test_score': test_metrics['r2']
        }

    stacking_path = models_dir / 'stacking_ensemble_phase95.joblib'
    save_model(stacking_model, str(stacking_path), metadata=stacking_metadata)
    print(f"\n✓ Stacking model saved: {stacking_path}")

    # Save quantile models
    for q, model in zip(QUANTILES, quantile_models):
        quantile_metadata = {
            'model_type': f'quantile_regressor_q{q}',
            'features': list(X_train_reg.columns),
            'target': target_col,
            'date_trained': datetime.now().strftime('%Y-%m-%d'),
            'phase': '9.5',
            'quantile': q
            }
        quantile_path = models_dir / f'quantile_q{int(q * 100)}_phase95.joblib'
        save_model(model, str(quantile_path), metadata=quantile_metadata)

    print(f"✓ Quantile models saved: {len(QUANTILES)} models")

    #%% md
    ### 9.5.8 Summary and Store Predictions

    #%%
    print_section_header("PHASE 9.5 IMPLEMENTATION SUMMARY")

    # Format best model description
    if best_model_name not in ["None", "N/A"] and not results_df.empty:
        best_model_summary = f"{best_model_name} (R²={results_df.loc[best_model_name, 'r2']:.4f})"
    else:
        best_model_summary = "Not available (comparison failed)"

    # Print summary
    classification_cols = [c for c in regression_df_clean.columns if c.startswith('event_prob_')]

    summary = {
        "✓ Classification Features Integrated": f"{len(classification_cols)} probability features + interactions",
        "✓ Models Compared": "6 models: Ridge, Lasso, RF, ET, GB, HistGB",
        "✓ Best Single Model": best_model_summary,
        "✓ Stacking Ensemble": f"R²={test_metrics['r2']:.4f}, MAE={test_metrics['mae']:.2f}",
        "✓ Quantile Regression": f"{len(QUANTILES)} quantiles for prediction intervals",
        "✓ Models Saved": f"{models_dir} (stacking + quantile models)"
        }

    for key, value in summary.items():
        print(f"\n{key}")
        print(f"  {value}")

    print(f"\n{SEPARATOR_LINE}")
    print("✓ Phase 9.5 Complete - Advanced Regression System Operational")
    print(SEPARATOR_LINE)

    # Store predictions for downstream use (Phases 9.6 and 9.7)
    if len(X_test_reg) > 0:
        test_indices = X_test_reg.index

        # Validate index alignment before assignment
        if 'all_stocks_featured' in globals():
            # Check which indices exist in all_stocks_featured
            valid_indices = test_indices.intersection(all_stocks_featured.index)
            missing_indices = len(test_indices) - len(valid_indices)

            if missing_indices > 0:
                print(f"  ⚠ Warning: {missing_indices} test indices not found in all_stocks_featured")

            if len(valid_indices) > 0:
                # Only assign to indices that exist
                all_stocks_featured.loc[valid_indices, 'predicted_price_target'] = y_pred_stacking[
                    test_indices.isin(valid_indices)
                ]
                all_stocks_featured.loc[valid_indices, 'prediction_lower_10'] = predictions_quantile[0.1][
                    test_indices.isin(valid_indices)
                ]
                all_stocks_featured.loc[valid_indices, 'prediction_upper_90'] = predictions_quantile[0.9][
                    test_indices.isin(valid_indices)
                ]

                print(f"\n✓ Predictions added to 'all_stocks_featured'")
                print(f"  Predictions added for {len(valid_indices):,} test samples")
            else:
                print(f"\n⚠ Warning: No matching indices found between X_test_reg and all_stocks_featured")

        # Create Phase 9.5 output dataset with validated indices
        all_stocks_phase95 = regression_df_enhanced.copy()
        valid_phase95_indices = test_indices.intersection(all_stocks_phase95.index)

        if len(valid_phase95_indices) > 0:
            all_stocks_phase95.loc[valid_phase95_indices, 'predicted_price_target'] = y_pred_stacking[
                test_indices.isin(valid_phase95_indices)
            ]
            all_stocks_phase95.loc[valid_phase95_indices, 'prediction_lower_10'] = predictions_quantile[0.1][
                test_indices.isin(valid_phase95_indices)
            ]
            all_stocks_phase95.loc[valid_phase95_indices, 'prediction_upper_90'] = predictions_quantile[0.9][
                test_indices.isin(valid_phase95_indices)
            ]

            print(f"✓ Dataset ready for Phase 9.6/9.7 (stored as 'all_stocks_phase95')")
            print(f"  {len(valid_phase95_indices):,} samples with predictions")

except Exception as e:
    logger.error(f"Phase 9.5 Advanced Regression failed: {e}")
    print(f"\n⚠ Phase 9.5 Advanced Regression failed: {e}")
    import traceback

    traceback.print_exc()
