def extract_feature_columns(df: pd.DataFrame) -> tuple:
    """Extract classification and valuation columns from dataframe."""
    classification_cols = [c for c in df.columns if c.startswith(CLASSIFICATION_COL_PREFIX)]
    valuation_cols = [c for c in VALUATION_COLUMNS if c in df.columns]
    return classification_cols, valuation_cols


def check_data_quality(df: pd.DataFrame, valuation_cols: list) -> None:
    """Check for data quality issues in valuation columns."""
    print("\n🔧 Checking for zero/infinite values...")
    for col in valuation_cols:
        if col in df.columns:
            zero_count = (df[col] == 0).sum()
            inf_count = np.isinf(df[col]).sum()
            if zero_count > 0 or inf_count > 0:
                print(f"  Warning: {col} has {zero_count} zeros and {inf_count} inf values")


def clean_missing_values(df: pd.DataFrame, target_col: str) -> pd.DataFrame:
    """Handle missing values in the dataset."""
    print("\n🔧 Handling missing values...")
    nan_before = df.isnull().sum().sum()
    print(f"  NaN values before cleanup: {nan_before}")

    if nan_before > 0:
        numeric_cols = df.select_dtypes(include=[np.number]).columns
        df = df.replace([np.inf, -np.inf], np.nan)

        for col in numeric_cols:
            if col != target_col and df[col].isnull().any():
                median_val = df[col].median()
                fill_value = median_val if pd.notna(median_val) else 0
                df[col] = df[col].fillna(fill_value)

        df = df.fillna(0)
        nan_after = df.isnull().sum().sum()
        print(f"  NaN values after cleanup: {nan_after}")

        if nan_after == 0:
            print("✓ No NaNs remaining - data ready for modeling")
        else:
            print(f"⚠ Warning: {nan_after} NaN values remain after cleanup")

    return df

def safe_train_models(train_func, *args, **kwargs) -> tuple:
    """Safely execute model training with comprehensive error handling."""
    try:
        return train_func(*args, **kwargs), None
    except ValueError as e:
        logger.warning(f"Model training failed due to data issues: {e}")
        print(f"\n⚠ Model training failed: {e}")
        print("  Check data quality and feature availability")
        return None, str(e)
    except RuntimeError as e:
        logger.warning(f"Model training runtime error: {e}")
        print(f"\n⚠ Model training failed: {e}")
        return None, str(e)
    except Exception as e:
        logger.error(f"Unexpected error in model training: {e}", exc_info=True)
        print(f"\n⚠ Model training failed: {e}")
        return None, str(e)


def create_predictions_dataframe(y_test, y_pred, predictions_quantile: dict) -> pd.DataFrame:
    """Create enhanced predictions dataframe with metadata."""
    predictions_df = pd.DataFrame({
        'y_true': y_test.values,
        'y_pred': y_pred,
        'residual': y_test.values - y_pred,
        'abs_error': np.abs(y_test.values - y_pred),
        'pct_error': ((y_test.values - y_pred) / y_test.values) * 100,
        'lower_10': predictions_quantile[QUANTILES[0]],
        'median': predictions_quantile[QUANTILES[1]],
        'upper_90': predictions_quantile[QUANTILES[2]]
        })
    return predictions_df


def validate_quantile_order(predictions_df: pd.DataFrame) -> pd.DataFrame:
    """Validate and correct quantile order (lower < median < upper)."""
    invalid_intervals = (
            (predictions_df['lower_10'] > predictions_df['median']) |
            (predictions_df['median'] > predictions_df['upper_90'])
    ).sum()

    if invalid_intervals > 0:
        print(f"  ⚠ Warning: {invalid_intervals} samples have invalid quantile order")
        print(f"    Correcting by sorting quantiles...")

        for idx in predictions_df.index:
            values = [
                predictions_df.loc[idx, 'lower_10'],
                predictions_df.loc[idx, 'median'],
                predictions_df.loc[idx, 'upper_90']
                ]
            sorted_values = sorted(values)
            predictions_df.loc[idx, 'lower_10'] = sorted_values[0]
            predictions_df.loc[idx, 'median'] = sorted_values[1]
            predictions_df.loc[idx, 'upper_90'] = sorted_values[2]

    return predictions_df

def save_stacking_model(model, models_dir: Path, test_metrics: dict,
                        stacking_results: dict, feature_cols: list, target_col: str) -> None:
    """Save stacking ensemble model with metadata."""
    from finance_ml.advanced_models import save_model

    stacking_metadata = {
        'model_type': 'stacking_ensemble',
        'features': feature_cols,
        'target': target_col,
        'date_trained': datetime.now().strftime('%Y-%m-%d'),
        'phase': '9.5',
        'train_score': stacking_results.get('train_score', 0),
        'cv_score': stacking_results.get('cv_score', 0),
        'test_score': test_metrics['r2']
        }

    stacking_path = models_dir / STACKING_MODEL_FILE
    save_model(model, str(stacking_path), metadata=stacking_metadata)
    print(f"\n✓ Stacking model saved: {stacking_path}")


def save_quantile_models(quantile_models: list, models_dir: Path,
                         feature_cols: list, target_col: str) -> None:
    """Save quantile regression models with metadata."""
    from finance_ml.advanced_models import save_model

    for q, model in zip(QUANTILES, quantile_models):
        quantile_metadata = {
            'model_type': f'quantile_regressor_q{q}',
            'features': feature_cols,
            'target': target_col,
            'date_trained': datetime.now().strftime('%Y-%m-%d'),
            'phase': '9.5',
            'quantile': q
            }
        quantile_path = models_dir / QUANTILE_MODEL_FILE_TEMPLATE.format(int(q * 100))
        save_model(model, str(quantile_path), metadata=quantile_metadata)

    print(f"✓ Quantile models saved: {len(QUANTILES)} models")

def create_sector_region_heatmap(all_stocks: pd.DataFrame) -> None:
    """Create and display sector-region market cap heatmap."""
    required_cols = ['sector', 'region', 'market_cap']

    if not all(col in all_stocks.columns for col in required_cols):
        print("\n⚠ Skipping sector-region heatmap: required columns not found")
        return

    try:
        heatmap_data = all_stocks.groupby(['sector', 'region'])['market_cap'].sum().unstack(fill_value=0)

        # Get top 10 sectors by total market cap
        top_sectors = heatmap_data.sum(axis=1).nlargest(10).index
        heatmap_data = heatmap_data.loc[top_sectors]

        if not heatmap_data.empty:
            plt.figure(figsize=(10, 8))
            sns.heatmap(heatmap_data / 1e9, annot=True, fmt='.1f', cmap='YlOrRd',
                        linewidths=0.5, cbar_kws={'label': 'Market Cap ($B)'})
            plt.title('Market Cap Distribution: Top 10 Sectors by Region',
                      fontsize=14, fontweight='bold')
            plt.xlabel('Region')
            plt.ylabel('Sector')
            plt.tight_layout()
            plt.show()
    except Exception as e:
        print(f"\n⚠ Failed to create heatmap: {e}")
