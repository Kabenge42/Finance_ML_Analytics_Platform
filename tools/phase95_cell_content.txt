def prepare_target_variable(df: pd.DataFrame, target_col: str, fallback_col: str) -> tuple[pd.DataFrame, str]:
    """Prepare target variable, using fallback if needed."""
    if target_col not in df.columns:
        print(f"\n⚠ Warning: '{target_col}' not found. Using '{fallback_col}' as proxy target.")
        target_col = fallback_col
    df_clean = df.dropna(subset=[target_col])
    print(f"\n✓ Clean dataset: {df_clean.shape[0]:,} rows")
    return df_clean, target_col


def extract_classification_columns(df: pd.DataFrame) -> list:
    """Extract classification probability columns from dataset."""
    return [c for c in df.columns if c.startswith('event_prob_')]


def extract_valuation_columns(df: pd.DataFrame) -> list:
    """Extract available valuation columns from dataset."""
    required_cols = ['p_e', 'p_b', 'ev_ebitda', 'market_cap']
    return [c for c in required_cols if c in df.columns]


def validate_interaction_prerequisites(classification_cols: list, valuation_cols: list) -> bool:
    """Validate that required columns exist for interaction features."""
    if not classification_cols or not valuation_cols:
        print("\n⚠ Skipping interaction features - missing required columns")
        return False
    print(f"\nClassification features: {len(classification_cols)}")
    print(f"Valuation features: {len(valuation_cols)}")
    return True


def create_interaction_features(df: pd.DataFrame) -> pd.DataFrame:
    """Create classification × valuation interaction features."""
    print_section_header("9.5.1 — Creating Classification Interaction Features")

    classification_cols = extract_classification_columns(df)
    valuation_cols = extract_valuation_columns(df)

    if not validate_interaction_prerequisites(classification_cols, valuation_cols):
        return df.copy()

    _check_valuation_quality(df, valuation_cols)
    df_enhanced = create_classification_interactions(
            df,
            classification_cols=classification_cols,
            valuation_cols=valuation_cols
            )

    _report_interaction_results(df, df_enhanced)
    return df_enhanced


def _report_interaction_results(df_original: pd.DataFrame, df_enhanced: pd.DataFrame) -> None:
    """Report on newly created interaction features."""
    interaction_cols = [c for c in df_enhanced.columns if '_x_' in c and c not in df_original.columns]
    print(f"\n✓ Created {len(interaction_cols)} interaction features")
    if interaction_cols[:3]:
        print(f"  Examples: {', '.join(interaction_cols[:3])}")


def _check_valuation_quality(df: pd.DataFrame, valuation_cols: list) -> None:
    """Check for zero/infinite values in valuation columns."""
    print("\n🔧 Checking for zero/infinite values before interaction...")
    for col in valuation_cols:
        if col in df.columns:
            zero_count = (df[col] == 0).sum()
            inf_count = np.isinf(df[col]).sum()
            if zero_count > 0 or inf_count > 0:
                print(f"  Warning: {col} has {zero_count} zeros and {inf_count} inf values")


def count_nan_values(df: pd.DataFrame) -> int:
    """Count total NaN values in dataframe."""
    return df.isnull().sum().sum()


def fill_numeric_columns(df: pd.DataFrame, target_col: str) -> pd.DataFrame:
    """Fill missing values in numeric columns with median or zero."""
    numeric_cols = df.select_dtypes(include=[np.number]).columns.tolist()
    numeric_cols_to_fill = [c for c in numeric_cols if c != target_col]

    for col in numeric_cols_to_fill:
        if df[col].isnull().any():
            median_val = df[col].median()
            fill_value = median_val if pd.notna(median_val) else 0
            df[col] = df[col].fillna(fill_value)
    return df


def handle_missing_values(df: pd.DataFrame, target_col: str) -> pd.DataFrame:
    """Handle missing and infinite values in enhanced dataset."""
    print("\n🔧 Handling missing values in enhanced dataset...")
    nan_before = count_nan_values(df)
    print(f"  NaN values before cleanup: {nan_before}")

    if nan_before == 0:
        return df

    df = fill_numeric_columns(df, target_col)
    df = df.fillna(0)
    df = df.replace([np.inf, -np.inf], np.nan).fillna(0)

    nan_after = count_nan_values(df)
    print(f"  NaN values after cleanup: {nan_after}")
    _report_cleanup_status(nan_after)
    return df


def _report_cleanup_status(nan_count: int) -> None:
    """Report on data cleanup status."""
    if nan_count == 0:
        print("✓ No NaNs remaining - data ready for modeling")
    else:
        print(f"⚠ Warning: {nan_count} NaNs still present")

def compare_regression_models(X_train, y_train, cv: int, random_state: int) -> pd.DataFrame:
    """Compare multiple regression models."""
    print_section_header("9.5.3 — Comparing Multiple Regression Models")

    try:
        comparison_results = compare_regressors(
                X_train, y_train,
                test_size=TEST_SIZE_RATIO,
                cv=cv,
                random_state=random_state,
                ensure_nonnegative=True
                )
        results_df = _format_comparison_results(comparison_results)
        print("\n📊 Model Comparison Results:")
        print(results_df.to_string())

    except Exception as e:
        return _handle_comparison_failure(e)


def _format_comparison_results(comparison_results: dict) -> pd.DataFrame:
    """Format and sort model comparison results."""
    return pd.DataFrame(comparison_results).T.sort_values('r2', ascending=False)


def _handle_comparison_failure(error: Exception) -> pd.DataFrame:
    """Handle model comparison failure."""
    print(f"\n⚠ Model comparison failed: {error}")

def train_stacking_ensemble_model(X_train, y_train, X_test, y_test, cv: int):
    """Train stacking ensemble and evaluate on test set."""
    print_section_header("9.5.4 — Training Stacking Ensemble")

    model, results = train_stacking_regressor(X_train, y_train, cv=cv, ensure_nonnegative=True)
    print_ensemble_training_info(results)

    y_pred = model.predict(X_test)
    test_metrics = _calculate_test_metrics(y_test, y_pred)
    print_test_metrics(test_metrics)

    return model, y_pred, test_metrics


def _calculate_test_metrics(y_true, y_pred) -> dict:
    """Calculate test set metrics."""
    from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score
    return {
        'mae': mean_absolute_error(y_true, y_pred),
        'rmse': np.sqrt(mean_squared_error(y_true, y_pred)),
        'r2': r2_score(y_true, y_pred)
        }


def get_quantile_score(q: float, model, results: dict, index: int, X_train, y_train) -> float:
    """Get training score for a quantile model."""
    try:
        if 'quantile_results' in results and isinstance(results['quantile_results'], list):
            if index < len(results['quantile_results']):
                return results['quantile_results'][index]['train_score']
        return model.score(X_train, y_train)
    except (KeyError, IndexError, TypeError):
        return model.score(X_train, y_train)


def print_quantile_models_info(quantiles: list, models: list, results: dict, X_train, y_train) -> None:
    """Print information about trained quantile models."""
    print(f"\n✓ Quantile Models Trained:")
    print(f"  Quantiles: {quantiles}")
    print(f"  Models: {len(models)}")

    for i, (q, model) in enumerate(zip(quantiles, models)):
        score = get_quantile_score(q, model, results, i, X_train, y_train)
        print(f"  Q{q}: {score:.4f} (train R²)")


def train_quantile_models(X_train, y_train, X_test, quantiles: list):
    """Train quantile regression models."""
    print_section_header("9.5.5 — Quantile Regression for Uncertainty Estimation")

    models, results = train_quantile_regressor(X_train, y_train, quantiles=quantiles)
    print_quantile_models_info(quantiles, models, results, X_train, y_train)

    predictions = {q: model.predict(X_test) for q, model in zip(quantiles, models)}
    return models, predictions


def create_predictions_dataframe(y_test, predictions_quantile: dict, y_pred_stacking) -> pd.DataFrame:
    """Create dataframe with predictions and intervals."""
    return pd.DataFrame({
        'actual': y_test.values,
        'lower_10': predictions_quantile[0.1],
        'median': predictions_quantile[0.5],
        'upper_90': predictions_quantile[0.9],
        'point_prediction': y_pred_stacking
        })


def plot_prediction_intervals(predictions_df: pd.DataFrame, sample_size: int) -> None:
    """Create plot showing prediction intervals."""
    sample_size = min(sample_size, len(predictions_df))
    sample_indices = range(sample_size)

    plt.figure(figsize=(14, 6))
    plt.fill_between(
            sample_indices,
            predictions_df['lower_10'][:sample_size],
            predictions_df['upper_90'][:sample_size],
            alpha=0.3, color='lightblue', label='80% Prediction Interval'
            )
    plt.plot(sample_indices, predictions_df['actual'][:sample_size],
             'o', color='black', label='Actual', markersize=4)
    plt.plot(sample_indices, predictions_df['median'][:sample_size],
             '-', color='blue', label='Median Prediction', linewidth=2)
    plt.plot(sample_indices, predictions_df['point_prediction'][:sample_size],
             '--', color='red', label='Stacking Prediction', linewidth=1.5)

    plt.xlabel('Sample Index')
    plt.ylabel('Price Target')
    plt.title('Prediction Intervals: Quantile Regression with Stacking Ensemble', fontweight='bold')
    plt.legend(loc='best')
    plt.grid(True, alpha=0.3)
    plt.tight_layout()
    plt.show()


def visualize_prediction_intervals(y_test, predictions_quantile: dict, y_pred_stacking,
                                   sample_size: int) -> pd.DataFrame:
    """Create and visualize prediction intervals."""
    predictions_df = create_predictions_dataframe(y_test, predictions_quantile, y_pred_stacking)

    print(f"\n📊 Prediction Intervals (first 5 samples):")
    print(predictions_df.head().to_string())

    plot_prediction_intervals(predictions_df, sample_size)
    return predictions_df


def determine_sort_metric(df: pd.DataFrame) -> str:
    """Determine best available metric for sorting sector results."""
    for metric in ['r2', 'train_score', 'mae']:
        if metric in df.columns:
            return metric
    print(f"\n⚠ Warning: None of the expected metric columns found.")
    print(f"   Available columns: {df.columns.tolist()}")
    return 'none'


def format_sector_summary(results: dict) -> pd.DataFrame:
    """Format sector model results as dataframe."""
    sector_metrics = results.get('sector_metrics', results)
    return pd.DataFrame(sector_metrics).T


def print_sector_summary(sector_summary: pd.DataFrame, sort_metric: str) -> None:
    """Print formatted sector model summary."""
    display_cols = [c for c in ['train_score', 'r2', 'mae', 'rmse', 'cv_mean', 'model_type']
                    if c in sector_summary.columns]

    if not sector_summary.empty and display_cols:
        print(sector_summary[display_cols].head(10).to_string())
    elif not sector_summary.empty:
        print(sector_summary.head(10).to_string())
    else:
        print("   No sector models trained (insufficient samples per sector)")


def train_sector_models(df: pd.DataFrame, X_train, target_col: str, min_samples: int) -> None:
    """Train sector-specific models if sector column exists."""
    print_section_header("9.5.6 — Sector-Specific Model Training")

    if 'sector' not in df.columns:
        print("\n⚠ Sector column not found - skipping sector-specific models")
        return

    feature_cols = list(X_train.columns)
    models, results = train_sector_specific_models(
            df,
            feature_cols=feature_cols,
            target_col=target_col,
            sector_col='sector',
            model_type='random_forest',
            min_samples=min_samples
            )

    print(f"\n✓ Sector-Specific Models Trained:")
    print(f"  Total sectors: {len(models)}")

    sector_summary = format_sector_summary(results)
    sort_metric = determine_sort_metric(sector_summary)

    if sort_metric != 'none':
        ascending = (sort_metric == 'mae')
        sector_summary = sector_summary.sort_values(sort_metric, ascending=ascending)

    print(f"\n📊 Sector Model Performance (sorted by {sort_metric}):")
    print_sector_summary(sector_summary, sort_metric)


def create_model_metadata(model_type: str, features: list, target_col: str,
                          phase: str = '9.5', **kwargs) -> dict:
    """Create metadata dictionary for model persistence."""
    from datetime import datetime

    metadata = {
        'model_type': model_type,
        'features': features,
        'target': target_col,
        'date_trained': datetime.now().strftime('%Y-%m-%d'),
        'phase': phase
        }
    metadata.update(kwargs)
    return metadata


def save_stacking_model(model, models_dir, X_train, target_col: str,
                        stacking_results: dict, test_r2: float) -> None:
    """Save stacking ensemble model with metadata."""
    metadata = create_model_metadata(
            model_type='stacking_ensemble',
            features=list(X_train.columns),
            target_col=target_col,
            train_score=stacking_results['train_score'],
            cv_score=stacking_results['cv_score'],
            test_score=test_r2
            )

    stacking_path = models_dir / 'stacking_ensemble_phase95.joblib'
    save_model(model, str(stacking_path), metadata=metadata)
    print(f"\n✓ Stacking model saved: {stacking_path}")


def save_quantile_models_batch(quantile_models: list, quantiles: list, models_dir,
                               X_train, target_col: str) -> None:
    """Save all quantile models with metadata."""
    for q, model in zip(quantiles, quantile_models):
        metadata = create_model_metadata(
                model_type=f'quantile_regressor_q{q}',
                features=list(X_train.columns),
                target_col=target_col,
                quantile=q
                )

        quantile_path = models_dir / f'quantile_q{int(q * 100)}_phase95.joblib'
        save_model(model, str(quantile_path), metadata=metadata)

    print(f"✓ Quantile models saved: {len(quantiles)} models")


def save_models_to_disk(stacking_model, quantile_models, quantiles: list, X_train,
                        target_col: str, test_r2: float, stacking_results: dict) -> None:
    """Save trained models with metadata."""
    print_section_header("9.5.7 — Model Persistence")

    from pathlib import Path
    models_dir = Path(config.output_dir) / 'models'
    models_dir.mkdir(exist_ok=True, parents=True)

    save_stacking_model(stacking_model, models_dir, X_train, target_col, stacking_results, test_r2)
    save_quantile_models_batch(quantile_models, quantiles, models_dir, X_train, target_col)
